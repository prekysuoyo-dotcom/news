// Function to handle Likes via GitHub async function handleLike() { if(!currentStory) return; const btn = document.getElementById('likeBtn'); btn.disabled = true; const resp = await fetch(SCRIPT_URL + "?action=like&row=" + currentStory.row); const newCount = await resp.text(); document.getElementById('countLikes').innerText = newCount; currentStory.likes = newCount; // Update local data btn.style.color = "var(--red)"; } // Function to handle Comments via GitHub async function postComment() { const name = document.getElementById('cName').value; const msg = document.getElementById('cMsg').value; if(!name || !msg) { alert("Please fill name and message"); return; } const btn = event.target; btn.innerText = "Posting..."; btn.disabled = true; // Use JSONP/Script injection for cross-domain writing const script = document.createElement('script'); script.src = `${SCRIPT_URL}?callback=commCallback&action=comment&slug=${currentStory.slug}&name=${encodeURIComponent(name)}&msg=${encodeURIComponent(msg)}`; document.body.appendChild(script); } // This clears the form after comment is sent window.commCallback = function(response) { alert("Comment Posted!"); document.getElementById('cName').value = ""; document.getElementById('cMsg').value = ""; // Reload comments view init(); };
